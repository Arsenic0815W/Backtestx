cmake_minimum_required(VERSION 3.20)
project(BacktestX VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(OBX_WERROR "Treat compiler warnings as errors" ON)
option(OBX_SANITIZE "Request sanitizers (not supported on MinGW; prints 'skipped')" OFF)

# --- Sanitizers -------------------------------------------------------------
# MinGW-w64 GCC does not ship libasan/libubsan, so we never enable them here.
# We say so out loud instead of silently doing nothing.
if(OBX_SANITIZE)
  if(MINGW)
    message(STATUS "OBX_SANITIZE: skipped (MinGW has no ASan/UBSan runtime)")
  else()
    message(STATUS "OBX_SANITIZE: skipped (only the MinGW build is supported in this project)")
  endif()
endif()

# --- Common warning flags ---------------------------------------------------
add_library(bt_warnings INTERFACE)
target_compile_options(bt_warnings INTERFACE -Wall -Wextra)
if(OBX_WERROR)
  target_compile_options(bt_warnings INTERFACE -Werror)
endif()

# Vendored headers: SYSTEM so their internal warnings never reach us.
add_library(bt_third_party INTERFACE)
target_include_directories(bt_third_party SYSTEM INTERFACE ${CMAKE_SOURCE_DIR}/third_party)

# --- Embedded report assets --------------------------------------------------
# src/report_assets/*.{html,css,js} are real files (editable, testable with Node) turned into a generated .cpp at configure time.
# file(READ)/file(WRITE) rather than configure_file(): CSS "@media" would collide with @VAR@ substitution. CRLF -> LF so the bytes
# never depend on how a file was saved. Editing an asset re-runs configure (CMAKE_CONFIGURE_DEPENDS).
set(BT_ASSET_DIR ${CMAKE_SOURCE_DIR}/src/report_assets)
set(BT_ASSET_CPP ${CMAKE_BINARY_DIR}/generated/report_assets.cpp)
set(_asset_code "// GENERATED by CMakeLists.txt from src/report_assets/ - do not edit.
#include \"bt/report_assets.hpp\"
namespace bt {
namespace report_assets {
")
foreach(_pair "report_template:report_template.html" "report_css:report.css" "report_js:report.js")
  string(REPLACE ":" ";" _pair "${_pair}")
  list(GET _pair 0 _fn)
  list(GET _pair 1 _file)
  file(READ "${BT_ASSET_DIR}/${_file}" _content)
  string(REPLACE "
" "
" _content "${_content}")
  string(FIND "${_content}" ")BTX_RAW\"" _clash)
  if(NOT _clash EQUAL -1)
    message(FATAL_ERROR "${_file} contains the raw-string terminator )BTX_RAW\"")
  endif()
  string(FIND "${_content}" "</script" _script_end)
  if(NOT _script_end EQUAL -1 AND NOT _file STREQUAL "report_template.html")
    message(FATAL_ERROR "${_file} contains '</script', which would end the inline script element")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BT_ASSET_DIR}/${_file}")
  string(APPEND _asset_code "const char* ${_fn}() { return R\"BTX_RAW(${_content})BTX_RAW\"; }
")
endforeach()
string(APPEND _asset_code "}  // namespace report_assets
}  // namespace bt
")
set(_old "")
if(EXISTS "${BT_ASSET_CPP}")
  file(READ "${BT_ASSET_CPP}" _old)
endif()
if(NOT _old STREQUAL _asset_code)
  file(WRITE "${BT_ASSET_CPP}" "${_asset_code}")
endif()

add_library(bt_core STATIC
  src/version.cpp
  src/date.cpp
  src/bar.cpp
  src/csv.cpp
  src/adjust.cpp
  src/calendar.cpp
  src/synthetic.cpp
  src/money.cpp
  src/costs.cpp
  src/fill.cpp
  src/portfolio.cpp
  src/engine.cpp
  src/sizing.cpp
  src/indicators.cpp
  src/strategies.cpp
  src/dataset.cpp
  src/analytics.cpp
  src/export.cpp
  src/run_config.cpp
  src/runner.cpp
  src/sweep.cpp
  src/report.cpp
  src/commands.cpp
  ${BT_ASSET_CPP}
)
target_include_directories(bt_core PUBLIC ${CMAKE_SOURCE_DIR}/include)
target_link_libraries(bt_core PUBLIC bt_third_party PRIVATE bt_warnings)

  tests/test_adjust.cpp
  tests/test_calendar.cpp

target_link_libraries(bt_bench PRIVATE bt_core bt_warnings)

add_executable(backtestx cli/main.cpp)
